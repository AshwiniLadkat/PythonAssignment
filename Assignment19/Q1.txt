#write a program which contains one lambda function which accepts one parameter and return 
#power of two
power_of_two = lambda x: x ** 2
is_int = lambda s: s.strip().lstrip("+-").isdigit()
def main():
    num = (input("Input: "))
    if is_int(num):        
        result = power_of_two(int(num))
        print(f"Output: {result}",end= ' ')
    else:
        print("Invalid input. Please enter an integer.")
if __name__ == "__main__":
    main()
