#write a program whihc contains  filter ,map and reduce in it.
#Python application which contains one list of numbers and accepted from user 
#  Filter - should filter numbers >= 70 and <= 90 
#  map - increase number by 10 
#  reduce - return pro duct number from that modified list

from functools import reduce    
def FilterNumbers(Task,Inputlist):
    result = list()
    for no in Inputlist:
        Ret = Task(no)
        if(Ret):
            result.append(no)
    return result
is_int = lambda s: s.strip().lstrip("+-").isdigit()
FilterRange = lambda A : True if A >= 70 and A <= 90 else False
IncreaseByTen = lambda A : A + 10
Multiply = lambda A,B : A * B
def main():   
    try:        
        inputdata = input("Input by comma separator: ").split(',')
        filterlist = list(FilterNumbers(is_int,inputdata))
        print("Data after  filter is : ",filterlist)
        numbers = list(map(int, filterlist))
        print("Data after  mapped is : ",numbers)
        filterlist2 = list(FilterNumbers(FilterRange,numbers))
        print("Data after  second filter is : ",filterlist2)
        modifiedlist = list(map(IncreaseByTen,filterlist2))
        print("Data after  second mapped is : ",modifiedlist)
        if len(modifiedlist) == 0:
            print("No valid integers were provided.")
            return
        result = reduce(Multiply,modifiedlist)             
        print("Product of numbers from the modified list is:", result)
    except ValueError:
        print("Invalid input. Please enter integers separated by spaces.")
    except Exception as e:
        print("An error occurred:", str(e))
if __name__ == "__main__":
    main()
