from functools import reduce
def FilterNumbers(Task,Inputlist):
    result = list()
    for no in Inputlist:
        Ret = Task(no)
        if(Ret):
            result.append(no)
    return result
def SearchNumber(Task,Inputlist):
    result = list()

        Ret = Task(no)

is_int = lambda s: s.strip().lstrip("+-").isdigit()
SearchNumber_Count =  lambda lst, n: lst.count(n)
def main():
    
    try:        
        inputdata = input("Input by comma separator: ").split(',')
        search_number = int(input("Enter the number to find its frequency: "))
        if(is_int(str(search_number))==False):
            print("Please enter a valid integer for frequency search.")
            return
        filterlist = list(FilterNumbers(is_int,inputdata))
        print("Data after  filter is : ",filterlist)
        numbers = list(map(int, filterlist))
        print("Data after  mapped is : ",numbers)
        if len(numbers) == 0:
            print("No valid integers were provided.")
            return
        result = numbers.count(search_number)             
        print(f"Frequency of {search_number} in the list is:", result)
    except ValueError:
        print("Invalid input. Please enter integers separated by spaces.")
    except Exception as e:
        print("An error occurred:", str(e))

if __name__ == "__main__":
