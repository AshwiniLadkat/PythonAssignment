#both the threads should accept a list of integers as imput from user.
#the even list thread should 
   #Extract all even elements from the list
    #Calculate  and display sum 
#the odd list thread should
   #Extract all odd elements from the list
    #Calculate  and display sum
#Thread should run concurrenly 

def even_thread(numbers):
    even_sum = sum(num for num in numbers if num % 2 == 0)
    print(f"Even numbers: {[num for num in numbers if num % 2 == 0]}")
    print(f"Sum of even numbers: {even_sum}")
def odd_thread(numbers):
    odd_sum = sum(num for num in numbers if num % 2 != 0)
    print(f"Odd numbers: {[num for num in numbers if num % 2 != 0]}")
    print(f"Sum of odd numbers: {odd_sum}")
import threading
def main():
    try:
        input_data = input("Enter a list of integers separated by spaces: ")
        numbers = list(map(int, input_data.split()))
        even_thread_obj = threading.Thread(target=even_thread, args=(numbers,))
        odd_thread_obj = threading.Thread(target=odd_thread, args=(numbers,))
        even_thread_obj.start()
        odd_thread_obj.start()
        even_thread_obj.join()
        odd_thread_obj.join()
    except ValueError:
        print("Invalid input. Please enter integers separated by spaces.")
    except Exception as e:
        print("An error occurred:", str(e))
if __name__ == "__main__":
    main()
