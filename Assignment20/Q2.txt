#Desingn a python application that creates two seperate threads namesd as EvenFactor and OddFactor.
#Both thread should accept a number from user as a parameter.
#EvenFctor- Identify the all even factors of that number and calculate and display 
            # sum of that even factors.
#Oddthread -Identify the all odd factors of that number and calculate and display 
            # sum of that odd factors.
# After both thread completes their execution, the main thread should display the message 
  #"End from main thread".
import threading
def even_factors_sum(number):
    print("Inside even_factors_sum:", threading.get_ident())
    even_sum = 0
    print(f"Even factors of {number}: ", end="")
    for i in range(1, number + 1):  
        if number % i == 0 and i % 2 == 0:
            print(i, end=" ")
            even_sum += i
    print(f"\nSum of even factors: {even_sum}")
def odd_factors_sum(number):
    print("Inside odd_factors_sum:", threading.get_ident())
    odd_sum = 0
    print(f"Odd factors of {number}: ", end="")
    for i in range(1, number + 1):  
        if number % i == 0 and i % 2 != 0:
            print(i, end=" ")
            odd_sum += i
    print(f"\nSum of odd factors: {odd_sum}")
def main():
    try:
        num = int(input("Enter a number: "))
        even_thread = threading.Thread(target=even_factors_sum, args=(num,))
        odd_thread = threading.Thread(target=odd_factors_sum, args=(num,))
        even_thread.start()
        odd_thread.start()
        even_thread.join()
        odd_thread.join()
        print("End from main thread")
    except ValueError:
        print("Invalid input. Please enter an integer.")

if __name__ == "__main__":
    main()
